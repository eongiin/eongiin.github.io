## 나의 풀이

```java
package test;

import org.junit.jupiter.api.Assertions;
import org.junit.jupiter.api.Test;

class Solution {
	static int max;
	static final int choice = 3;

	public int solution(int n, int m, int[] card) {
		max = -1;
		select(0, 0, n, m, card, new int[choice]);
		return max;
	}

	void select(int cnt, int idx, int n, int m, int[] card, int[] myCard) {
		if (cnt == choice) {
			int sum = 0;
			for (int num : myCard) {
				sum += num;
			}

			if (sum > m)
				return;
			max = Math.max(max, sum);
			return;
		}

		for (int i = idx; i < n; i++) {
			myCard[cnt] = card[i];
			select(cnt + 1, i + 1, n, m, card, myCard);
		}
	}

	@Test
	void test() {
		Assertions.assertEquals(21, solution(5, 21, new int[] { 5, 6, 7, 8, 9 }));
		Assertions.assertEquals(497, solution(10, 500, new int[] { 93, 181, 245, 214, 315, 36, 185, 138, 216, 295 }));
	}

}
```

## 최종 코드

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {
	static int max;
	static final int choice = 3;

	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

		StringTokenizer st = new StringTokenizer(br.readLine());
		int n = Integer.parseInt(st.nextToken());
		int m = Integer.parseInt(st.nextToken());

		st = new StringTokenizer(br.readLine());
		int[] card = new int[n];
		int idx = 0;
		while (st.hasMoreTokens()) {
			card[idx++] = Integer.parseInt(st.nextToken());
		}
		
		max = -1;

		select(0, 0, n, m, card, new int[choice]);
		System.out.println(max);
	}

	static void select(int cnt, int idx, int n, int m, int[] card, int[] myCard) {
		if (cnt == choice) {
			int sum = 0;
			for (int num : myCard) {
				sum += num;
			}

			if (sum > m)
				return;
			max = Math.max(max, sum);
			return;
		}

		for (int i = idx; i < n; i++) {
			myCard[cnt] = card[i];
			select(cnt + 1, i + 1, n, m, card, myCard);
		}
	}

}
```
