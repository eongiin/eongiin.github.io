## 나의 풀이

```java
package test;

import org.junit.jupiter.api.Assertions;
import org.junit.jupiter.api.Test;

class Solution {
	static int min;
	static int[][] ability;

	public int solution(int n, int[][] ability) {
		min = Integer.MAX_VALUE;
		Solution.ability = ability;
		int[] member = new int[n];
		for (int i = 0; i < n; i++)
			member[i] = i;
		makeTeam(0, 0, n, new boolean[n]);
		return min;
	}

	static void makeTeam(int cnt, int idx, int n, boolean[] selected) {
		if (cnt == n / 2) {
			int result = calc(n, selected);
			min = Math.min(min, result);
			return;
		}

		for (int i = idx; i < n; i++) {
			selected[i] = true;
			makeTeam(cnt + 1, i + 1, n, selected);
			selected[i] = false;
		}
	}

	static int calc(int n, boolean[] selected) {
		int sumA = 0;
		int sumB = 0;
		for (int i = 0; i < n - 1; i++) {
			for (int j = i + 1; j < n; j++) {
				if (selected[i] && selected[j]) {
					sumA += ability[i][j];
					sumA += ability[j][i];
				} else if (!selected[i] && !selected[j]) {
					sumB += ability[i][j];
					sumB += ability[j][i];
				}
			}
		}
		return Math.abs(sumA - sumB);
	}

	@Test
	void test() {
		Assertions.assertEquals(0,
				solution(4, new int[][] { { 0, 1, 2, 3 }, { 4, 0, 5, 6 }, { 7, 1, 0, 2 }, { 3, 4, 5, 0 } }));
		Assertions.assertEquals(2, solution(6, new int[][] { { 0, 1, 2, 3, 4, 5 }, { 1, 0, 2, 3, 4, 5 },
				{ 1, 2, 0, 3, 4, 5 }, { 1, 2, 3, 0, 4, 5 }, { 1, 2, 3, 4, 0, 5 }, { 1, 2, 3, 4, 5, 0 } }));
		Assertions.assertEquals(1,
				solution(8,
						new int[][] { { 0, 5, 4, 5, 4, 5, 4, 5 }, { 4, 0, 5, 1, 2, 3, 4, 5 },
								{ 9, 8, 0, 1, 2, 3, 1, 2 }, { 9, 9, 9, 0, 9, 9, 9, 9 }, { 1, 1, 1, 1, 0, 1, 1, 1 },
								{ 8, 7, 6, 5, 4, 0, 3, 2 }, { 9, 1, 9, 1, 9, 1, 0, 9 }, { 6, 5, 4, 3, 2, 1, 9, 0 } }));
	}

}
```
처음엔 한번 계산한 팀 조합은 다시 안 하게 하려고  
(teamA가 0,1 teamB가 2,3 인 상황과 teamA가 2,3 teamB가 0,1 인 상황의 차이는 똑같으니까 )  
set에 넣고 그 배열을 포함하고 있으면 계산 안 하게 하려 했는데  
배열은 그렇게 비교할 수가 없어서 그냥 list로 바꾸고 list를 차례로 돌면서 Arrays.equals(a, b)로 비교했다  
근데 이렇게 하는 코드를 넣으니까 오히려 시간이 더 걸려서 뺐다  
그리고 teamA teamB를 따로 만들어서 아래처럼 계산도 해봤으나 메모리만 많이 차지하고 시간은 크게 차이 없길래 그냥 위의 방식으로 했다

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

import java.util.Arrays;

public class Main {

	static int[][] ability;

		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		int n = Integer.parseInt(br.readLine());
		int[][] ability = new int[n][n];
		for (int i = 0; i < n; i++) {
			StringTokenizer st = new StringTokenizer(br.readLine());
			for (int j = 0; j < n; j++) {
				ability[i][j] = Integer.parseInt(st.nextToken());
			}
		}

		Main.ability = ability;
		min = Integer.MAX_VALUE;

		makeTeam(0, 0, n, new boolean[n]);
		System.out.println(min);
	}

	static void makeTeam(int cnt, int idx, int n, boolean[] selected) {

			selected[i] = true;

		}
	}

	static int calc(int n, boolean[] selected) {

	}
}
