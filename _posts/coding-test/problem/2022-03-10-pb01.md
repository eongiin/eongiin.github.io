```java
package test;

import org.junit.jupiter.api.Assertions;
import org.junit.jupiter.api.Test;

class Solution {
	static int max = -1;
	static boolean[] chk;

	public int solution(int n, int[] a) {
		chk = new boolean[n];

		setOrder(0, n, new int[6], a);
		return max;
	}

	void setOrder(int cnt, int n, int[] making, int[] a) {
		if (cnt == n) {
			int result = calc(making);
			max = Math.max(max, result);
			return;
		}

		for (int i = 0; i < n; i++) {
			if (chk[i])
				continue;
			chk[i] = true;
			making[cnt] = a[i];
			setOrder(cnt + 1, n, making, a);
			chk[i] = false;
		}
	}

	int calc(int[] array) {
		int sum = 0;
		for (int i = 0; i < array.length - 1; i++) {
			sum += Math.abs(array[i] - array[i + 1]);
		}
		return sum;
	}

	@Test
	void test() {
		Assertions.assertEquals(62, solution(6, new int[] { 20, 1, 15, 8, 4, 10 }));
	}

}
```

```java
import java.io.*;
import java.util.*;
public class Main {
	static int n;
	static int max = Integer.MIN_VALUE;
    public static void main(String[] args) throws IOException {
       Scanner sc = new Scanner(System.in);
       n = sc.nextInt();
       int[] arr = new int[n];
       for(int i = 0 ; i<n; i++) {
    	   arr[i] = sc.nextInt();
       }
       permutation(arr, 0);
       System.out.println(max);
    
    }
    static void permutation(int[] arr, int depth) {
    	if(depth == n) {
    		cal(arr);
    		return;
    	}
    	
    	for(int i = depth; i<n; i++) {
    		swap(arr, i, depth);
    		permutation(arr, depth+1);
    		swap(arr, i, depth);
    	}
    }
    static void cal(int[] arr) {
    	int sum = 0;
    	for(int i = 0 ; i<arr.length-1; i++) {
    		sum += Math.abs(arr[i+1] - arr[i]);
    	}
    	if(max < sum)
    		max = sum;
    }
    static void swap(int[] arr, int num, int depth) {
    	int tmp = arr[num];
    	arr[num] = arr[depth];
    	arr[depth] = tmp;
    }
}
```
