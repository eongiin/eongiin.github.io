## 나의 풀이


```java
import org.junit.jupiter.api.Assertions;
import org.junit.jupiter.api.Test;

class Solution {
	static int min;
	static int max;
	static int[] number;

	public int[] solution(int n, int[] number, int[] oper) {
		min = Integer.MAX_VALUE;
		max = Integer.MIN_VALUE;
		Solution.number = number;

		makeOrder(0, n - 1, oper, new int[n - 1]);

		return new int[] { max, min };
	}

	void makeOrder(int cnt, int n, int[] oper, int[] making) {
		if (cnt == n) {

			int result = calc(making);
			min = Math.min(min, result);
			max = Math.max(max, result);
			return;
		}

		for (int i = 0; i < oper.length; i++) {
			if (oper[i]==0)
				continue;
			oper[i] -= 1;
			making[cnt] = i;
			makeOrder(cnt + 1, n, oper,making);
			oper[i] += 1;
		}
	}

	int calc(int[] made) {
		int result = number[0];

		for (int i = 0; i < made.length; i++) {
			int idx = made[i];

			switch (idx) {
			case 0:
				result += number[i + 1];
				break;
			case 1:
				result -= number[i + 1];
				break;
			case 2:
				result *= number[i + 1];
				break;
			case 3:
				result /= number[i + 1];
				break;
			}
		}
		return result;
	}

	@Test
	void test() {
		Assertions.assertArrayEquals(new int[] { 30, 30 }, solution(2, new int[] { 5, 6 }, new int[] { 0, 0, 1, 0 }));
		Assertions.assertArrayEquals(new int[] { 35, 17 },
				solution(3, new int[] { 3, 4, 5 }, new int[] { 1, 0, 1, 0 }));
		Assertions.assertArrayEquals(new int[] { 54, -24 },
				solution(6, new int[] { 1, 2, 3, 4, 5, 6 }, new int[] { 2, 1, 1, 1 }));

	}

}
```
처음 주어지는 숫자 정보 - number[], 연산자 타입별 개수 정보 - oper[]  

makeOrder 함수에서 oper[] 인덱스 정보를 이용하여 연산자들의 순서를 만든다  
oper[] 에 접근해 값이 0이 아니라면 그 인덱스를 선택하고 1을 뺀다   
함수를 부르고 나서는 다시 되돌려 놓는다  
그렇게 making[] 을 완성하면 calc함수에서 oper[] 인덱스에 해당하는 값으로 계산을 한다  


## 최종 코드

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;
import java.util.StringTokenizer;

public class Main {

	static int min;
	static int max;
	static int[] number;

	public static void main(String[] args) throws IOException {

		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		int n = Integer.parseInt(br.readLine());

		min = Integer.MAX_VALUE;
		max = Integer.MIN_VALUE;
		number = new int[n];

		StringTokenizer st = new StringTokenizer(br.readLine(), " ");
		for (int i = 0; i < n; i++) {
			number[i] = Integer.parseInt(st.nextToken());
		}

		st = new StringTokenizer(br.readLine(), " ");
		int[] oper = new int[4];
		for (int i = 0; i < oper.length; i++) {
			oper[i] = Integer.parseInt(st.nextToken());
		}

		perm(number[0], 1, n, oper);

		System.out.println(max);
		System.out.println(min);

	}

	static void perm(int num, int idx, int n, int[] oper) {
		if (idx == n) {
			max = Math.max(max, num);
			min = Math.min(min, num);
			return;
		}

		for (int i = 0; i < oper.length; i++) {
			if (oper[i] > 0) {
				oper[i]--;
				switch (i) {
				case 0:
					perm(num + number[idx], idx + 1, n, oper);
					break;
				case 1:
					perm(num - number[idx], idx + 1, n, oper);
					break;
				case 2:
					perm(num * number[idx], idx + 1, n, oper);
					break;
				case 3:
					perm(num / number[idx], idx + 1, n, oper);
					break;
				}
				oper[i]++;
			}
		}
	}

}
```
