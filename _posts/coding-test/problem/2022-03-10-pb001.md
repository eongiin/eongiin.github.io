## 나의 풀이

```java
package test;

import org.junit.jupiter.api.Assertions;
import org.junit.jupiter.api.Test;

class Solution {
	static int max;
	static boolean[] chk;

	public int solution(int n, int[] a) {
		max = -1;
		chk = new boolean[n];

		setOrder(0, n, new int[n], a);
		return max;
	}

	void setOrder(int cnt, int n, int[] making, int[] a) {
		if (cnt == n) {
			int result = calc(making);
			max = Math.max(max, result);
			return;
		}

		for (int i = 0; i < n; i++) {
			if (chk[i])
				continue;
			chk[i] = true;
			making[cnt] = a[i];
			setOrder(cnt + 1, n, making, a);
			chk[i] = false;
		}
	}

	int calc(int[] array) {
		int sum = 0;
		for (int i = 0; i < array.length - 1; i++) {
			sum += Math.abs(array[i] - array[i + 1]);
		}
		return sum;
	}

	@Test
	void test() {
		Assertions.assertEquals(62, solution(6, new int[] { 20, 1, 15, 8, 4, 10 }));
	}

}
```

## 최종 코드

~~~java

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {
	static int max;
	static boolean[] chk;

	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		int n = Integer.parseInt(br.readLine());

		StringTokenizer st = new StringTokenizer(br.readLine());
		int[] a = new int[n];
		int idx = 0;
		while (st.hasMoreTokens()) {
			a[idx++] = Integer.parseInt(st.nextToken());
		}
		max = -1;
		chk = new boolean[n];

		setOrder(0, n, new int[n], a);
		System.out.println(max);
	}

	static void setOrder(int cnt, int n, int[] making, int[] a) {
		if (cnt == n) {
			int result = calc(making);
			max = Math.max(max, result);
			return;
		}

		for (int i = 0; i < n; i++) {
			if (chk[i])
				continue;
			chk[i] = true;
			making[cnt] = a[i];
			setOrder(cnt + 1, n, making, a);
			chk[i] = false;
		}
	}

	static int calc(int[] array) {
		int sum = 0;
		for (int i = 0; i < array.length - 1; i++) {
			sum += Math.abs(array[i] - array[i + 1]);
		}
		return sum;
	}

}
~~~
