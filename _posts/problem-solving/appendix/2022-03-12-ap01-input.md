## Scanner vs BufferedReader

| | Scanner | BufferedReader |
| ------ | -------- | ---------- |
| Buffer Size | 1024 |8192|
| Syncronized | x | o|
| 문자열 파싱| 가능| 단순히 읽어 들임|
|IOException|숨김|던짐|

BufferedReader 가 단순히 읽어들여서 속도가 더 빠르다

### 예시

#### Scanner
```java
package test;

import java.util.Scanner;


public class Simple {

	public static void main(String[] args) throws IOException {
		
		Scanner sc = new Scanner(System.in);
        int intValue = sc.nextInt();
        sc.nextLine();//엔터키 소진
        System.out.println("int "+intValue);
        
        long longValue = sc.nextLong();
        sc.nextLine();
        System.out.println("long "+longValue);
        
        double doubleValue = sc.nextDouble();
        sc.nextLine();
        System.out.println("double "+doubleValue);
        
        
        String strValue = sc.next();
        sc.nextLine();
        System.out.println("str1 "+strValue);
        
        String strValue2=sc.nextLine();
        System.out.println("str2 "+strValue2);
        
        boolean boolValue = sc.nextBoolean();
        sc.nextLine();
        System.out.println("bool "+boolValue);
	}
}
```
#### BufferedReader

```java
package test;

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Simple {

	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

		String s = br.readLine();
		System.out.println("String : " + s);

		// int
		int number = Integer.parseInt(br.readLine());
		System.out.println("int : " + number);

		// 여러개

		// split
		String[] s2 = br.readLine().split(" ");
		for (int i = 0; i < s2.length; i++) {
			System.out.println("String " + (i + 1) + ": " + s2[i]);
		}

		// StringTokenizer
		StringTokenizer st = new StringTokenizer(br.readLine());
		int[] s3 = new int[st.countTokens()];
		int count = 0;
		while (st.hasMoreTokens()) {
			s3[count] = Integer.parseInt(st.nextToken());
			System.out.print(s3[count++] + " ");
		}
		System.out.println();

		// 2차원배열 (숫자)

		st = new StringTokenizer(br.readLine());
		int n = Integer.parseInt(st.nextToken());
		int m = Integer.parseInt(st.nextToken());

		int[][] graph = new int[n][m];

		for (int i = 0; i < n; i++) {
			st = new StringTokenizer(br.readLine());
			for (int j = 0; j < m; j++) {
				graph[i][j] = Integer.parseInt(st.nextToken());
				System.out.print(graph[i][j] + " ");
			}
			System.out.println();
		}

		// 2차원배열 (문자), 공백없이주어진경우
		st = new StringTokenizer(br.readLine());
		int n2 = Integer.parseInt(st.nextToken());
		int m2 = Integer.parseInt(st.nextToken());

		char[][] arr = new char[n2][m2];
		for (int i = 0; i < n2; i++) {
			st = new StringTokenizer(br.readLine());
			arr[i] = st.nextToken().toCharArray();
			System.out.println(arr[i]);
		}
	}
}
```
