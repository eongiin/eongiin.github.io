```java
String sql = "select " + buildDistQuery(latitude, longitude) + " as dist, wi.*" 
+ " from wifi_info wi"
+ " where lat not null and lnt not null" + " order by dist" + " limit 20;";
```

```java
private String buildDistQuery(double lat, double lnt) {

  final double myCosLat = Math.cos(Math.toRadians(lat));
  final double myCosLnt = Math.cos(Math.toRadians(lnt));
  final double mySinLat = Math.sin(Math.toRadians(lat));
  final double mySinLnt = Math.sin(Math.toRadians(lnt));
  
  return "6371*acos(" + myCosLat + "*cos_lat*(cos_lnt*" + myCosLnt + "+sin_lnt*" + mySinLnt + ")+" + mySinLat + "*sin_lat) ";
}
```
