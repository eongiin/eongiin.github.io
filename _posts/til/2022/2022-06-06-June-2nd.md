#### Dijkstra
- 출발점에서 목표점까지의 최단 경로를 구하는 알고리즘
- 한 노드에서 다른 모든 노드로의 최단 경로 구함
- 간선에 음의 가중치가 없어야 함
- 그리디 + DP 
- O(ElogV)

##### 인접 리스트
- 양방향이면 양쪽 모두 넣어야 하는 것 주의

```java
import java.util.ArrayList;
import java.util.PriorityQueue;

public class Solution {

    static final int INF = 1000_000_000;
    ArrayList<Node>[] graph;
    int[] dist;
    boolean[] visited;

    public void solution(int n, int e, int[][] arr, int start) {

        graph = new ArrayList[n + 1];
        dist = new int[n + 1];
        visited = new boolean[n + 1];

        for (int i = 1; i < n + 1; i++) {
            graph[i] = new ArrayList<>();
            dist[i] = INF;
        }

        for (int i = 0; i < e; i++) {
            graph[arr[i][0]].add(new Node(arr[i][1], arr[i][2]));
            graph[arr[i][1]].add(new Node(arr[i][0], arr[i][2]));
        }

        dijkstra(start);

        for (int i = 1; i < n + 1; i++) {
            if (dist[i] == INF) {
                System.out.print("INF ");
            } else {
                System.out.print(dist[i] + " ");
            }
        }
    }

    void dijkstra(int start) {

        PriorityQueue<Node> pq = new PriorityQueue<>();
        pq.offer(new Node(start, 0));
        dist[start] = 0;

        while (!pq.isEmpty()) {
            Node cur = pq.poll();

            if (visited[cur.num]) {
                continue;
            }
            visited[cur.num] = true;

            if (dist[cur.num] < cur.cost) {
                continue;
            }

            for (Node adj : graph[cur.num]) {
                int newCost = cur.cost + adj.cost;
                if (!visited[adj.num] && dist[adj.num] > newCost) {
                    dist[adj.num] = newCost;
                    pq.offer(new Node(adj.num, newCost));
                }
            }
        }
    }

    class Node implements Comparable<Node> {
        int num;
        int cost;

        public Node(int num, int cost) {
            this.num = num;
            this.cost = cost;
        }

        @Override
        public int compareTo(Node o) {
            return this.cost - o.cost;
        }
    }
}
```
##### 인접 행렬
- 양방향이면 양쪽 모두 넣어야 하는 것 주의
- 한 노드에서 다른 노드로 바로 가는 방법이 여러 개인 경우 최솟값으로 저장해 두어야 한다

```java
int[][] graph;
int[] distance;

void solution(int v, int[][] data, int start) {
	graph = new int[v + 1][v + 1];
	distance = new int[v + 1];

	for (int i = 0; i < data.length; i++) {
		graph[data[i][0]][data[i][1]] = data[i][2];
		graph[data[i][1]][data[i][0]] = data[i][2];
	}

	for (int i = 1; i <= v; i++) {
		distance[i] = Integer.MAX_VALUE;
	}

	dijkstra(start);
}

void dijkstra(int start) {
	distance[start] = 0;
	PriorityQueue<Node> pq = new PriorityQueue<>((x, y) -> x.cost - y.cost);
	pq.offer(new Node(start, 0));

	while (!pq.isEmpty()) {
		Node cur = pq.poll();

		if (distance[cur.num] < cur.cost) {
			continue;
		}

		for (int adjNum = 1; adjNum < graph.length; adjNum++) {
			int adjCost = graph[cur.num][adjNum];

			if (adjCost == 0) {
				continue;
			}

			int newCost = cur.cost + adjCost;

			if (newCost < distance[adjNum]) {
				distance[adjNum] = newCost;
				pq.offer(new Node(adjNum, newCost));
			}
		}
	}
}

class Node {
	int num;
	int cost;

	public Node(int num, int cost) {
		this.num = num;
		this.cost = cost;
	}
}
```
#### Bellman-Ford
- 음수 간선이 포함되어 있어도 최단 경로 구할 수 있음
	- 음수 사이클이 있으면 정상 동작 X
- 매번 모든 간선을 확인하기 때문에 다익스트라에 비해 느림
- O(VE)

```java
Edge[] edges;
int[] distance;
boolean isMinusCycle;

void solution(int v, int e, int[][] data, int start) {

	edges = new Edge[e];
	for (int i = 0; i < e; i++) {
		edges[i] = new Edge(data[i][0], data[i][1], data[i][2]);
	}

	distance = new int[v + 1];
	for (int i = 1; i <= v; i++) {
		distance[i] = Integer.MAX_VALUE;
	}

	bellmanFord(v, e, start);
}

void bellmanFord(int v, int e, int start) {

	distance[start] = 0;
	isMinusCycle = false;

	//사이클 검사하려고 v+1
	for (int i = 0; i < v + 1; i++) {
		for (int j = 0; j < e; j++) {
			Edge cur = edges[j];

			if (distance[cur.from] == Integer.MAX_VALUE) {
				continue;
			}

			if (distance[cur.to] > (distance[cur.from] + cur.cost)) {
				distance[cur.to] = distance[cur.from] + cur.cost;

				if (i == v) {
					isMinusCycle = true;
				}
			}
		}
	}
}

class Edge {
	int from;
	int to;
	int cost;

	public Edge(int from, int to, int cost) {
		this.from = from;
		this.to = to;
		this.cost = cost;
	}
}
```

#### Floyd-Warshall

- 모든 노드 간의 최단 경로를 구하는 알고리즘
- 음의 간선이 포함되어 있어도 사용 가능
	- 음수 사이클이 있으면 정상 동작 X
- O(V^3)

```java
int[][] distance;
static final int INF = (int) 1e9;

void solution(int v, int e, int[][] data) {
	distance = new int[v + 1][v + 1];
	for (int i = 1; i <= v; i++) {
		for (int j = 1; j <= v; j++) {
			if (i != j) {
				distance[i][j] = INF;
			}
		}
	}

	for (int i = 0; i < e; i++) {
		distance[data[i][0]][data[i][1]] = data[i][2];
	}
	floydWarshall();
}

void floydWarshall() {
	int v = distance.length - 1;

	for (int k = 1; k <= v; k++) {
		for (int i = 1; i <= v; i++) {
			for (int j = 1; j <= v; j++) {
				if (distance[i][k] != INF && distance[k][j] != INF) {
					distance[i][j] = Math.min(distance[i][j], distance[i][k] + distance[k][j]);
				}
			}
		}
	}
}
```

### 최소 신장 트리

- MST : Minimum Spanning Tree
- 그래프 상의 모든 노드들을 최소 비용으로 연결하는 방법
- 크루스칼, 프림

#### Kruskal
- 간선 중 최소 값을 가진 간선부터 연결
- 사이클 발생 시 다른 간선 선택
- 주로 간선 수가 적을 때 사용
- O(ElogE) (간선 sorting)

```java
int[] parents;

int kruskal(int[][] data, int v, int e) {
	int weightSum = 0;

	Arrays.sort(data, (x, y) -> x[2] - y[2]);

	parents = new int[v + 1];
	for (int i = 1; i < v + 1; i++) {
		parents[i] = i;
	}

	for (int i = 0; i < e; i++) {
		if (find(data[i][0]) != find(data[i][1])) {
			union(data[i][0], data[i][1]);
			weightSum += data[i][2];
		}
	}

	return weightSum;
}


int find(int a) {
	if (a != parents[a]) {
		parents[a] = find(parents[a]);
	}
	return parents[a];
}

void union(int a, int b) {
	int aP = find(a);
	int bP = find(b);

	if (aP < bP) {
		parents[bP] = aP;
	} else {
		parents[aP] = bP;
	}
}
```

#### Prim
- 임의의 노드에서 시작
- 연결된 노드의 간선 중 낮은 가중치를 갖는 간선 선택
- 간선의 개수가 많을 때 크루스칼보다 유리
- O(ElogV)

```java
import java.util.ArrayList;
import java.util.PriorityQueue;

public class Solution {

    ArrayList<Node>[] graph;
    boolean[] visited;

    public int solution(int n, int e, int[][] arr) {

        graph = new ArrayList[n + 1];
        visited = new boolean[n + 1];

        for (int i = 1; i < n + 1; i++) {
            graph[i] = new ArrayList<>();
        }
        for (int i = 0; i < e; i++) {
            graph[arr[i][0]].add(new Node(arr[i][1], arr[i][2]));
            graph[arr[i][1]].add(new Node(arr[i][0], arr[i][2]));
        }
        return prim();
    }

    int prim() {

        PriorityQueue<Node> pq = new PriorityQueue<>();
        pq.offer(new Node(1, 0));//임의의 노드

        int cnt = 0;
        int costSum = 0;
        while (!pq.isEmpty()) {
            Node cur = pq.poll();

            if (visited[cur.num]) {
                continue;
            }
            cnt++;
            visited[cur.num] = true;
            costSum += cur.cost;

            if (cnt == graph.length - 1) {
                return costSum;
            }

            for (Node adj : graph[cur.num]) {
                if (!visited[adj.num]) {
                    pq.offer(adj);
                }
            }
        }
        return costSum;
    }

    class Node implements Comparable<Node> {
        int num;
        int cost;

        public Node(int num, int cost) {
            this.num = num;
            this.cost = cost;
        }

        @Override
        public int compareTo(Node o) {
            return this.cost - o.cost;
        }
    }
}
```

#### 17-2 (Bellman Ford)

- n 개의 우주, m 개의 포탈, w 개의 웜홀
- 각 포탈과 웜홀에는 시간 비용이 있는데 웜홀을 통해 이동하는 경우는 시간이 거꾸로 흐른다
- 음수 사이클이 발생하는지 판별

```java
final static int INF = (int) 1e9;
Edge[] edge;
int[] dist;

void solution(int n, int m, int w, int[][] portal, int[][] wormhole) {

	edge = new Edge[m + w];
	dist = new int[n + 1];

	for (int i = 0; i < m; i++) {
		edge[i] = new Edge(portal[i][0], portal[i][1], portal[i][2]);
	}
	for (int i = 0; i < w; i++) {
		edge[m + i] = new Edge(wormhole[i][0], wormhole[i][1], -wormhole[i][2]);
	}

	for (int i = 0; i < n + 1; i++) {
		dist[i] = INF;
	}

	dist[1] = 0;
	System.out.println(bellmanFord(n, m + w));

}

boolean bellmanFord(int v, int e) {
	boolean isMinusCycle = false;

	for (int i = 0; i < v + 1; i++) {

		for (int j = 0; j < e; j++) {
			Edge cur = edge[j];

			if (dist[cur.from] == INF) {
				continue;
			}

			if (dist[cur.to] > dist[cur.from] + cur.weight) {
				dist[cur.to] = dist[cur.from] + cur.weight;

				if (i == v) {
					isMinusCycle = true;
				}
			}
		}
	}
	return isMinusCycle;
}

class Edge {
	int from;
	int to;
	int weight;

	public Edge(int from, int to, int weight) {
		this.from = from;
		this.to = to;
		this.weight = weight;
	}
}
```

#### 17-3 (Floyd Warshall)

- n 개의 도시, m 개의 버스
- 모든 도시의 쌍 (A,B)에 대해 도시 A에서 B로 가는데 필요한 비용의 최솟값 구하기
- 시작 도시와 도착 도시를 연결하는 노선은 하나가 아닐 수 있다
- 갈 수 없는 경로는 0 출력

```java
int[][] dist;
static final int INF = (int) 1e9;

void solution(int city, int bus, int[][] busInfo) {

	dist = new int[city + 1][city + 1];

	for (int i = 1; i < city + 1; i++) {
		for (int j = 1; j < city + 1; j++) {
			if (i != j) {
				dist[i][j] = INF;
			}
		}
	}

	for (int i = 0; i < bus; i++) {
		dist[busInfo[i][0]][busInfo[i][1]] = Math.min(dist[busInfo[i][0]][busInfo[i][1]], busInfo[i][2]);
	}

	floydWarshall(city);

	for (int i = 1; i < city + 1; i++) {
		for (int j = 1; j < city + 1; j++) {

			if (dist[i][j] >= INF) {
				System.out.printf("%5d", 0);
			} else {
				System.out.printf("%5d", dist[i][j]);
			}
		}
		System.out.println();
	}
}

void floydWarshall(int v) {
	for (int k = 1; k < v + 1; k++) {
		for (int j = 1; j < v + 1; j++) {
			for (int i = 1; i < v + 1; i++) {

				if (dist[i][k] != INF && dist[k][j] != INF) {
					dist[i][j] = Math.min(dist[i][j], dist[i][k] + dist[k][j]);
				}
			}
		}
	}
}
```
